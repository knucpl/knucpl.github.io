<script>
  (function () {
    function init() {
      var root = document.getElementById('kms-voyage');
      if (!root) return;
      var cards = root.querySelectorAll('.voyage-card');
      var bgs = root.querySelectorAll('.voyage-bg');
      var dotsBox = document.getElementById('kms-voyage-dots');
      var n = cards.length;
      var active = 0;
      var dir = 1;
      var dots = [];

      for (var k = 0; k < n; k++) {
        (function (k) {
          var b = document.createElement('button');
          b.type = 'button';
          b.setAttribute('aria-label', 'Show photo ' + (k + 1));
          b.addEventListener('click', function () { go(k); });
          dotsBox.appendChild(b);
          dots.push(b);
        })(k);
      }

      function setPos(card, p) {
        var prev = card._pos;
        // A hidden card switching sides jumps there instantly instead of sweeping across
        if (prev !== undefined && Math.abs(p - prev) > 2) {
          card.style.transition = 'none';
          card.dataset.pos = p > 0 ? '2' : '-2';
          void card.offsetWidth;
          card.style.transition = '';
        }
        card.dataset.pos = String(p);
        card._pos = p;
      }

      function render() {
        for (var i = 0; i < n; i++) {
          var d = ((i - active) % n + n) % n;
          if (d > n / 2 || (d === n / 2 && dir > 0)) d -= n;
          if (d > 2) d = 2;
          if (d < -2) d = -2;
          setPos(cards[i], d);
          cards[i].setAttribute('aria-hidden', d === 0 ? 'false' : 'true');
          bgs[i].classList.toggle('is-active', d === 0);
          dots[i].classList.toggle('is-active', d === 0);
        }
      }

      function go(i) {
        var target = ((i % n) + n) % n;
        if (target === active) return;
        var d = ((target - active) % n + n) % n;
        dir = d <= n / 2 ? 1 : -1;
        active = target;
        render();
      }
      function next() { dir = 1; active = (active + 1) % n; render(); }
      function prev() { dir = -1; active = (active - 1 + n) % n; render(); }

      root.querySelector('.voyage-next').addEventListener('click', next);
      root.querySelector('.voyage-prev').addEventListener('click', prev);

      Array.prototype.forEach.call(cards, function (card) {
        card.addEventListener('click', function () {
          if (moved) return;
          if (card._pos === 1) next();
          else if (card._pos === -1) prev();
        });
      });

      root.addEventListener('keydown', function (e) {
        if (e.key === 'ArrowRight') { next(); e.preventDefault(); }
        else if (e.key === 'ArrowLeft') { prev(); e.preventDefault(); }
      });

      // Swipe / drag
      var startX = null, startY = 0, moved = false;
      root.addEventListener('pointerdown', function (e) {
        if (e.target.closest('.voyage-nav')) return;
        startX = e.clientX;
        startY = e.clientY;
        moved = false;
      });
      root.addEventListener('pointerup', function (e) {
        if (startX === null) return;
        var dx = e.clientX - startX;
        var dy = e.clientY - startY;
        startX = null;
        if (Math.abs(dx) > 40 && Math.abs(dx) > Math.abs(dy)) {
          moved = true;
          if (dx < 0) next(); else prev();
          setTimeout(function () { moved = false; }, 0);
        }
      });
      root.addEventListener('pointercancel', function () { startX = null; });

      render();
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', init);
    } else {
      init();
    }
  })();
</script>

<script>
  (function () {
    var PHOTO_BASE = "{{ '/assets/img/people/' | relative_url }}";

    function init() {
      var names = document.querySelectorAll('.person[data-person]');
      if (!names.length) return;
      if (window.matchMedia && !window.matchMedia('(hover: hover)').matches) return;

      var card = document.createElement('div');
      card.id = 'person-hover-card';
      var img = document.createElement('img');
      img.alt = '';
      card.appendChild(img);
      document.body.appendChild(card);

      var CARD_W = 84, CARD_H = 92, GAP = 14;

      function move(e) {
        var dx = (e.clientX + CARD_W + GAP > window.innerWidth) ? -(CARD_W + GAP) : GAP;
        var dy = (e.clientY + CARD_H + GAP > window.innerHeight) ? -(CARD_H + GAP) : GAP;
        card.style.transform = 'translate(' + (e.clientX + dx) + 'px,' + (e.clientY + dy) + 'px)';
      }

      names.forEach(function (el) {
        el.addEventListener('mouseenter', function (e) {
          img.src = PHOTO_BASE + el.dataset.person + '.png';
          card.style.opacity = '1';
          move(e);
        });
        el.addEventListener('mousemove', move);
        el.addEventListener('mouseleave', function () {
          card.style.opacity = '0';
        });
      });
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', init);
    } else {
      init();
    }
  })();
</script>
