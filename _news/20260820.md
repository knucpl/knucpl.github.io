- Our group took part in the **KPOP-MS 2026** field campaign, organized by Kyungpook National University
- The campaign aims to observe mesoscale convective systems that develop over Gyeonggi Bay and move into the Seoul metropolitan area
- The intensive observation period (IOP) ran for three weeks, from July 29 to August 18, 2026
- During the IOP, [StormTracker](https://doi.org/10.5194/amt-13-5395-2020) sondes were launched every 3 hours at four sites — more than 170 launches at each site
- About 100 people from ten institutions contributed to the campaign:
    - Kyungpook National University (lead)
    - Pukyong National University
    - Seoul National University
    - Yonsei University
    - Colorado State University
    - Florida State University
    - Environment and Climate Change Canada
    - National Taiwan University
    - National Central University
    - Chinese Culture University
- Our group contributed at the campaign headquarters and at the Muui-do Island site, one of the four launch sites
    - **Headquarters**: <span class="person" data-person="kwonil">Kwonil</span> (weeks 2–3, operations lead) and <span class="person" data-person="seri">Seri</span> (weeks 1–2, weather briefing and instrument monitoring)
    - **Muui-do Island**: <span class="person" data-person="hyeonsik">HyeonSik</span> (weeks 1–2), <span class="person" data-person="jungmin">Jungmin</span> (week 2), <span class="person" data-person="seeun">Seeun</span> (week 3), and <span class="person" data-person="seokhyun">Seokhyun</span> (week 3)

  (function () {
    var PHOTO_BASE = "{{ '/assets/img/people/' | relative_url }}";

    function init() {
      var names = document.querySelectorAll('.person[data-person]');
      if (!names.length) return;
      if (window.matchMedia && !window.matchMedia('(hover: hover)').matches) return;

      var card = document.createElement('div');
      card.id = 'person-hover-card';
      var img = document.createElement('img');
      img.alt = '';
      card.appendChild(img);
      document.body.appendChild(card);

      var CARD_W = 84, CARD_H = 92, GAP = 14;

      function move(e) {
        var dx = (e.clientX + CARD_W + GAP > window.innerWidth) ? -(CARD_W + GAP) : GAP;
        var dy = (e.clientY + CARD_H + GAP > window.innerHeight) ? -(CARD_H + GAP) : GAP;
        card.style.transform = 'translate(' + (e.clientX + dx) + 'px,' + (e.clientY + dy) + 'px)';
      }

      names.forEach(function (el) {
        el.addEventListener('mouseenter', function (e) {
          img.src = PHOTO_BASE + el.dataset.person + '.png';
          card.style.opacity = '1';
          move(e);
        });
        el.addEventListener('mousemove', move);
        el.addEventListener('mouseleave', function () {
          card.style.opacity = '0';
        });
      });
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', init);
    } else {
      init();
    }
  })();
</script>
